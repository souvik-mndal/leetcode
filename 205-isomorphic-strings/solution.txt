class Solution {
public:
    bool isIsomorphic(string s, string t) {
        vector<int>mp(256,0);
        vector<bool>chk(256,0);
        for( int i=0 ; i<s.size() ; i++ ){
            if( mp[s[i]]==0 && chk[t[i]]==0 ){
                mp[s[i]] = t[i];
                chk[t[i]]=1;
            }
        }
        for( int i=0 ; i<s.size() ; i++ ){
            if( mp[s[i]] != t[i] ){
                return false;
            }
        }
        return true;
    }
};
